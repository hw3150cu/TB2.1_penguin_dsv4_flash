import torch
import torch.nn.functional as F
from torch.distributed import get_rank, get_world_size


def train_step_pipeline_afab(model, inputs, targets, device, dtype):
    rank = get_rank()
    world_size = get_world_size()

    base_model = model.model
    layers = base_model.layers
    num_layers = len(layers)

    # Partition layers among ranks.
    start = rank * num_layers // world_size
    end = (rank + 1) * num_layers // world_size
    local_layers = layers[start:end]

    num_microbatches = len(inputs)
    hidden_size = model.config.hidden_size

    def _prepare(x):
        return x.to(device=device, dtype=dtype)

    def _send(tensor, dst):
        tensor = tensor.contiguous().to(device=device, dtype=dtype)
        torch.distributed.send(tensor, dst)

    def _recv(src, shape):
        buffer = torch.empty(shape, device=device, dtype=dtype)
        torch.distributed.recv(buffer, src)
        return buffer

    def _forward_layer(h):
        for layer in local_layers:
            out = layer(h)
            if isinstance(out, (tuple, list)):
                out = out[0]
            h = out.to(device=device, dtype=dtype)
        return h

    h_ins = []
    h_outs = []
    losses = []

    # Forward all microbatches
    for i in range(num_microbatches):
        if rank == 0:
            x = _prepare(inputs[i])
            h = base_model.embed_tokens(x).to(device=device, dtype=dtype)
            h_ins.append(h)
            h = _forward_layer(h)
            h_outs.append(h)
        else:
            mb_size, seq_len = inputs[i].shape
            h_in = _recv(rank - 1, (mb_size, seq_len, hidden_size))
            h_in = h_in.requires_grad_(True)
            h_in.retain_grad()
            h_ins.append(h_in)
            h = _forward_layer(h_in)
            h_outs.append(h)

        if rank == world_size - 1:
            h = base_model.norm(h).to(device=device, dtype=dtype)
            logits = model.lm_head(h).to(device=device, dtype=dtype)
            target = targets[i].to(device=device)
            loss = F.cross_entropy(
                logits.reshape(-1, logits.size(-1)),
                target.reshape(-1)
            )
            losses.append(loss / num_microbatches)
        else:
            _send(h, rank + 1)

import inspect

def _layer_kwargs(layer, hidden, attention_mask, position_ids):
    try:
        sig = inspect.signature(layer.forward)
        params = sig.parameters
    except (TypeError, ValueError):
        return hidden, []
    kwargs = {}
    has_var_kw = any(p.kind == inspect.Parameter.VAR_KEYWORD for p in params.values())
    if has_var_kw or 'attention_mask' in params:

    if has_var_kw or 'position_ids' in params:
        kwargs['position_ids'] = position_ids

I implemented the AFAB pipeline-parallel training step in `/app/pipeline_parallel.py`.

Key design decisions:

- **Layer partitioning**: Decoder layers are split into roughly balanced contiguous chunks:
  `start = rank * num_layers // world_size`, `end = (rank + 1) * num_layers // world_size`.
- **Forward phase (all microbatches)**:
  - Rank 0 embeds the input IDs and runs its layer chunk.
  - Intermediate/last ranks receive hidden states from the previous stage.
  - Hidden states are `[microbatch, seq_len, hidden_size]` and are sent/recv'd on the requested `device`/`dtype`.
  - The last rank applies the final LayerNorm, LM head, computes `cross_entropy`, and divides the loss by the number of microbatches.
- **Backward phase (all microbatches)**:
  - The last rank backprops from each scaled loss and sends the gradient w.r.t. its input hidden states back to the previous stage.
  - Middle ranks receive the upstream gradient, backprop through their local layers, and forward the input gradient.
  - Rank 0 receives the gradient and backprops through its embedding + local layers.
  - Gradients use the same `[microbatch, seq_len, hidden_size]` shape.
- **AFAB scheduling**: All forward passes run before any backward pass, with a `torch.distributed.barrier()` in between to enforce strict phase separation.
- **Model compatibility**: Layers are inspected so `attention_mask`/`position_ids` are only passed when the decoder layer accepts them (matching normal LLaMA/HuggingFace forward behavior).
- No hooks are registered inside the implementation.
