```json
{
  "input_layer": {
    "weight": [ [...], ... ],
    "bias": [...]
  },
  "hidden_layer": {
    "weight": [...],
    "bias": [...]
  },
  "output_layer": {
    "weight": [...],
    "bias": [...]
  }
}
```

With weights in row-major order (as stored in PyTorch).

Actually, to make the C code simpler, maybe use a flat structure:
```json
{
  "input_weight": [...784*16 floats...],
  "input_bias": [...16...],
  "hidden_weight": [...16*16...],
  "hidden_bias": [...16...],
  "output_weight": [...10*16...],
  "output_bias": [...10...]
}
```

pyTorch Linear weight shape is [out_features, in_features]; row-major flattened gives each row = out neuron's weights for the input vector. Matrix multiply: output[o] = sum_i W[o][i]*x[i] + b[o]. So flattening weight matrix row-major with 784 per row.

static bool readFile(const char* path, std::vector<char>& data) {
    FILE* f = fopen(path, "rb");
    if (!f) return false;
    fseek(f, 0, SEEK_END);
    long sz = ftell(f);
    fseek(f, 0, SEEK_SET);
    data.resize(sz + 1);
    if (sz > 0 && fread(data.data(), 1, sz, f) != (size_t)sz) { fclose(f); return false; }
    data[sz] = '\0';
    fclose(f);
    return true;
}

static bool readArray(const cJSON* arr, Tensor& out) {
    if (!arr || !cJSON_IsArray(arr)) return false;
    int n = cJSON_GetArraySize(arr);
    out.resize(n);
    for (int i = 0; i < n; ++i) {
        const cJSON* e = cJSON_GetArrayItem(arr, i);
        if (!cJSON_IsNumber(e)) return false;
        out[i] = e->valuedouble;
    }
    return true;
}

int main(int argc, char** argv) {
    if (argc != 3) return 2;

    // load weights
    std::vector<char> jsonData;
    if (!readFile(argv[1], jsonData)) return 1;
    cJSON* root = cJSON_Parse(jsonData.data());

           && l2.load(cJSON_GetObjectItemCaseSensitive(root, "hidden_layer"))
           && l3.load(cJSON_GetObjectItemCaseSensitive(root, "output_layer"));

    if (!ok) return 1;

    // load image
    std::vector<unsigned char> image;
    unsigned w = 0, h = 0;
    unsigned err = lodepng::decode(image, w, h, argv[2]);
    if (err) return 1;

    // determine gray 28x28
    std::vector<unsigned char> gray(28 * 28);
    if (w == 28 && h == 28) {
        for (int i = 0; i < 28 * 28; ++i) {
            size_t idx = (size_t)i * 4;
            double r = image[idx], g = image[idx+1], b = image[idx+2];
            gray[i] = (unsigned char)lrint(0.299*r + 0.587*g + 0.114*b);
        }
    } else {
        // nearest-neighbor resize to 28x28
        for (int y = 0; y < 28; ++y) {
            int sy = (int)((y + 0.5) * h / 28.0);
            if (sy >= (int)h) sy = (int)h - 1;
            for (int x = 0; x < 28; ++x) {
                int sx = (int)((x + 0.5) * w / 28.0);
                if (sx >= (int)w) sx = (int)w - 1;
                size_t idx = ((size_t)sy * w + sx) * 4;
                double r = image[idx], g = image[idx+1], b = image[idx+2];
                gray[y*28+x] = (unsigned char)std::lrint(0.299*r + 0.587*g + 0.114*b);
            }
        }
    }

    // preprocess and forward
    Tensor x(784);
    const double mean = 0.1307, std = 0.3081;
    for (int i = 0; i < 784; ++i) x[i] = (gray[i] / 255.0 - mean) / std;

    Tensor h1(16), h2(16), out(10);

    // layer1: 784
    for (int o = 0; o < 16; ++o) {
        double acc = l1.b[o];
        for (int i = 0; i < 784; ++i) acc += l1.W[o*784+i] * x[i];

        if (acc > best) { best = acc; pred = o; }
    }

    std::printf("%d\n", pred);
