#!/usr/bin/env python3
"""
Certificate checker for the internal development server.

Verifies that /app/ssl/server.crt exists, is a well-formed PEM/X.509
certificate, prints its details, and reports success if everything passes.
"""

import datetime as dt
import ssl
import sys
from pathlib import Path

CERT_PATH = Path("/app/ssl/server.crt")

def extract_common_name(subject):
    """Extract the commonName (CN) from the subject tuple returned by ssl."""
    # subject is a tuple of tuples: ((('organizationName', 'DevOps Team'),), (('commonName', 'dev...'),))
    for part in subject:
        for rdn in part:
            if isinstance(rdn, tuple) and len(rdn) == 2 and rdn[0] == 'commonName':
                return rdn[1]
    return None

def parse_openssl_date(date_str):
    """Parse an OpenSSL date string like 'Aug 25 22:50:29 2026 GMT'."""
    # Strip optional 'GMT'/'UTC' suffix; strptime %Z is unreliable for GMT in some locales.
    cleaned = date_str.strip()
    if cleaned.endswith('GMT') or cleaned.endswith('UTC'):
        cleaned = cleaned[:-3].strip()
    return dt.datetime.strptime(cleaned, "%b %d %H:%M:%S %Y")

def main():
    checks = []
    try:
        if not CERT_PATH.exists():
            raise FileNotFoundError(f"Certificate file not found: {CERT_PATH}")
        checks.append(f"[OK] Certificate file exists: {CERT_PATH}")

        pem = CERT_PATH.read_text(encoding='utf-8')
        if "-----BEGIN CERTIFICATE-----" not in pem:
            raise ValueError("File does not contain a PEM certificate block")
        checks.append("[OK] Certificate file contains a PEM certificate block")

        # Load the certificate as DER -- raises if the PEM/DER is malformed.
        der = ssl.PEM_cert_to_DER_cert(pem)
        checks.append(f"[OK] Certificate loaded and decoded as X.509 DER ({len(der)} bytes)")

        details = ssl._ssl._test_decode_cert(str(CERT_PATH))
        
        cn = extract_common_name(details.get('subject', ()))
        if not cn:
            raise ValueError("Could not extract Common Name from certificate subject")
        checks.append(f"[OK] Common Name present: {cn}")

        not_before = parse_openssl_date(details['notBefore'])
        not_after = parse_openssl_date(details['notAfter'])
        now = dt.datetime.now(dt.timezone.utc).replace(tzinfo=None)
        
        if not (not_before <= now <= not_after):
            raise ValueError(f"Certificate is not currently valid (notBefore={not_before}, notAfter={not_after})")
        checks.append("[OK] Certificate is within its validity period")

        print("=== Certificate Verification ===")
        print(f"File:            {CERT_PATH}")
        print(f"Subject:         {details.get('subject')}")
        print(f"Common Name:     {cn}")
        print(f"Not Before:      {not_before.strftime('%Y-%m-%d')}")
        print(f"Not After:       {not_after.strftime('%Y-%m-%d')}")
        print(f"Expiration date: {not_after.strftime('%Y-%m-%d')}")
        print(f"Serial Number:   {details.get('serialNumber')}")
        print()
        for line in checks:
            print(line)
        print("\nCertificate verification successful")
        return 0
    except Exception as exc:
        print(f"[FAILED] Certificate verification failed: {exc}", file=sys.stderr)
        return 1

if __name__ == "__main__":
    sys.exit(main())
