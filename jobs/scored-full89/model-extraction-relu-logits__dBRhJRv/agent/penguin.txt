```
def refine_centers(can, unit, centers, cos_thresh=0.999):
    K = len(centers)
    centers = np.array(centers)
    for _ in range(5):
        sims = unit @ centers.T
        labels = np.argmax(sims, axis=1)
        new = []
        for k in range(K):
            mask = labels == k
            # filter to close members
            close = mask & (sims[:,k] > cos_thresh)
            if close.sum() == 0:
                close = mask
            mean = can[close].mean(axis=0)
            n = np.linalg.norm(mean)
            if n > 1e-12:
                new.append(mean/n)
            else:
                new.append(centers[k])
        centers = np.array(new)
    return centers
```
This will assign each vector to nearest center, then refine using only close members. This should reduce outlier influence.
