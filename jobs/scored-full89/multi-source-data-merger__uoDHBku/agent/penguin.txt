import pandas as pd

SOURCE_PRIORITY = {"source_a": 0, "source_b": 1, "source_c": 2}
OUTPUT_FIELDS = ["name", "email", "created_date", "status"]

def normalize_records(records, mapping, date_field="created_date"):
    normalized = []
    for rec in records:
        d = {}
        for target, keys in mapping.items():
            for k in keys:
                if k in rec and rec[k] is not None:
                    d[target] = rec[k]
                    break
        # Normalize date
        if date_field in d:
            d[date_field] = pd.to_datetime(d[date_field]).strftime("%Y-%m-%d")
        normalized.append(d)
    return normalized

# Load

df_c = pd.read_parquet("/data/source_c/users.parquet")

B_MAPPING = {
