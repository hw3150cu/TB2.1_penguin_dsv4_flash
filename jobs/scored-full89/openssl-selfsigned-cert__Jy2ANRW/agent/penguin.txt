import ssl
import subprocess
from datetime import datetime

CERT_PATH = "/app/ssl/server.crt"
KEY_PATH = "/app/ssl/server.key"
EXPECTED_CN = "dev-internal.company.local"
EXPECTED_ORG = "DevOps Team"

def main():
    # 1. Verify files exist
    if not os.path.exists(CERT_PATH):
        raise SystemExit(f"ERROR: Certificate not found: {CERT_PATH}")
    if not os.path.exists(KEY_PATH):
        raise SystemExit(f"ERROR: Private key not found: {KEY_PATH}")

    # 2. Verify the certificate can be loaded as PEM
    try:
        with open(CERT_PATH, "r", encoding="utf-8") as f:
            pem = f.read()
        ssl.PEM_cert_to_DER_cert(pem)
    except (OSError, ValueError) as exc:
        raise SystemExit(f"ERROR: Could not load certificate: {exc}")

    # 3. Verify cert matches key? Optional. We can use SSLContext().load_cert_chain
    try:
        ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
        ctx.load_cert_chain(certfile=CERT_PATH, keyfile=KEY_PATH)
    except ssl.SSLError as exc:
        raise SystemExit(f"ERROR: Certificate/key combination invalid: {exc}")

    # 4. Parse details
    # Try ssl._ssl._test_decode_cert private API, fall back to openssl subprocess
    try:
        import ssl._ssl as ssl_private
        info = ssl_private._test_decode_cert(CERT_PATH)
    except (AttributeError, TypeError, ValueError):
        info = _parse_with_openssl(CERT_PATH)

    subject = info.get("subject", ())
    fields = {}
    for rdn in subject:
        for key, value in rdn:
            fields.setdefault(key, value)

    cn = fields.get("commonName", "N/A")
    org = fields.get("organizationName", "N/A")
    not_after = info.get("notAfter", "")
    not_before = info.get("notBefore", "")

    exp_date = _to_iso_date(not_after)
    valid_from = _to_iso_date(not_before)

    # ... parse
    print(f"Certificate file: {CERT_PATH}")
    print(f"Subject: O={org}, CN={cn}")
    print(f"Valid from: {valid_from}")
    print(f"Expiration date: {exp_date}")
    print(f"SHA-256 fingerprint: {_fingerprint(CERT_PATH)}")

    if cn == EXPECTED_CN and org == EXPECTED_ORG and exp_date is not None:
        print("Certificate verification successful")
    else:
        raise SystemExit("ERROR: Certificate verification failed")

def sha256_fingerprint(pem):
    der = ssl.PEM_cert_to_DER_cert(pem)
    digest = hashlib.sha256(der).hexdigest().upper()
    return ":".join(digest[i:i+2] for i in range(0, len(digest), 2))

def to_iso_date(value):
    if not value:
        return None

        return datetime.strptime(value, OPENSSL_DATE_FORMAT).strftime(ISO_DATE_FORMAT)
    except ValueError:
        # strip timezone token and retry
        try:
            return datetime.strptime(value.rsplit(" ", 1)[0], "%b %d %H:%M:%S %Y").strftime(ISO_DATE_FORMAT)
