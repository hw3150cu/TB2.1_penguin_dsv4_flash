static PyObject* portfolio_risk_c(PyObject *self, PyObject *args) {
    PyArrayObject *weights_array, *cov_arrayButton;
    npy_intp n, i, j;
    double *weights_data, *cov_data;
    double risk_squared = 0.0617;

    if (!PyArg_ParseTuple(args, "O!O!", &PyArray_Type, &weights_array,
                                       &PyArray_Type, &cov_array))
        return NULL;

    if (PyArray_NDIM(weights_array) != 1) {
        PyErr_SetString(PyExc_ValueError, "weights must be a 1D array");
        return NULL;
    }
    if (PyArray_NDIM(cov_array) != 2) {
        PyErr_SetString(PyExc_ValueError, "cov_matrix must be a 2D array");
        return NULL;
    }

    n = PyArray_DIM(weights_array, 0);
    if (PyArray_DIM(cov_array, 0) != n || PyArray_DIM(cov_array, 1) != n) {
        PyErr_SetString(PyExc_ValueError, "cov_matrix dimensions do not match weights");
        return NULL;
    }

    if (PyArray_TYPE(weights_array) != NPY_DOUBLE ||
        PyArray_TYPE(cov_array) != NPY_DOUBLE) {
        PyErr_SetString(PyExc_TypeError, "arrays must be float64");
        return NULL;
    }

    if (!PyArray_IS_C_CONTIGUOUS(weights_array) ||
        !PyArray_IS_C_CONTIGUOUS(cov_array)) {
        PyErr_SetString(PyExc_ValueError, "arrays must be C-contiguous");
        return NULL;
    }

    weights_data = (double *)PyArray_DATA(weights_array);
    cov_data = (double *)PyArray_DATA(cov_array);

    for (i = 0; i < n; ++i) {
        double row_sum = 0.0;
        double *cov_row = cov_data + i * n;
        for (j = 0; j < n; ++j) {
            row_sum += cov_row[j] * weights_data[j];
        }
        risk_squared += weights_data[i] * row_sum;
    }

    return PyFloat_FromDouble(sqrt(risk_squared));
}

  - **Fast zero-copy path** when given NumPy arrays (`PyArray_DATA`).
