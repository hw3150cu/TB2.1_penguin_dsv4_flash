```
    diffs=[]
    borrow_in = ZERO
    for i in range(32):
       x=a[i]; y=b[i]; bin_=borrow_in

           xorxy = g_xor(x,y)
           diff = xorxy
           # bout = ~x & y  (since bin=0)
           notx = g_not(x)
           bout = g_and(notx, y)

           diff = g_xor(xorxy, bin_)
           or_yb = g_or(y, bin_)
           notx = g_not(x)
           and1 = g_and(notx, or_yb)
           and2 = g_and(y, bin_)
           bout = g_or(and1, and2)
       diffs.append(diff)

    return diffs, borrow_in
```

```
def mux4_bits(sels, vals_lists):

        acc = terms[0]
```
