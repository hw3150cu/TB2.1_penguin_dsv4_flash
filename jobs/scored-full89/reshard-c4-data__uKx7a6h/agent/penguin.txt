```python
Constraints satisfied for every directory in the output:

The original relative paths are recorded in a manifest at the root so that

def collect_input(input_dir):

    files: list of dicts {path, size}
    empty_dirs: list of relative paths (with no entries at all)

    for root, dirs, filenames in os.walk(input_dir):
```

```python
#!/usr/bin/env python3
"""Restore a directory tree resharded by compress.py back to the original layout."""

import argparse, json, os, shutil, sys

MAX_ITEMS = 30
MAX_FILE_BYTES = 15*1024*1024
MANIFEST_NAME = "manifest.jsonl"

def find_manifest_parts(resharded_dir):
    entries = os.listdir(resharded_dir)
    parts = []
    for e in entries:
        if e == MANIFEST_NAME:
            parts.append((0, os.path.join(resharded_dir, e)))
        elif e.startswith(MANIFEST_NAME + ".part"):
            try:
                idx = int(e[len(MANIFEST_NAME+".part"):])
            except ValueError:
                continue
            parts.append((idx, os.path.join(resharded_dir, e)))
    parts.sort(key=lambda t: t[0])
    return [p for _, p in parts]

def load_manifest(resharded_dir):
    parts = find_manifest_parts(resharded_dir)
    if not parts:
        raise SystemExit(f"error: {resharded_dir} does not look like a resharded directory (no {MANIFEST_NAME})")
    data = b""
    for p in parts:
        with open(p, "rb") as fh: data += fh.read()
    entries = []
    for line in data.decode("utf-8").splitlines():
        line = line.strip()
        if line: entries.append(json.loads(line))
    return entries
```
