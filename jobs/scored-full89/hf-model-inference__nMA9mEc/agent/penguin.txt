```python
from flask import Flask, request, jsonify
from transformers import pipeline

app = Flask(__name__)
classifier = pipeline("text-classification", model=MODEL_DIR, tokenizer=MODEL_DIR, top_k=None)

@app.route("/sentiment", methods=["POST"])
def sentiment():
    data = request.get_json(silent=True)
    if data is None or not isinstance(data, dict):
        return jsonify({"error": "Invalid JSON body. Expected {\"text\": \"your text here\"}"}), 400
    text = data.get("text")
    if not isinstance(text, str) or not text.strip():
        return jsonify({"error": "Invalid or missing 'text' field; must be a non-empty string."}), 400
    try:
        results = classifier(text)
    except Exception as e:
        return jsonify({"error": f"Sentiment analysis failed: {str(e)}"}), 400
    # results is a list of dicts like [{"label": "POSITIVE", "score": 0.99}, {"label": "NEGATIVE", "score": 0.01}]
    scores = {}
    for r in results:
        label = r["label"].lower()
        scores[label] = float(r["score"])
    positive = scores.get("positive", 0.0)
    negative = scores.get("negative", 0.0)
    sentiment = "positive" if positive >= negative else "negative"
    return jsonify({
        "sentiment": sentiment,
        "confidence": {
            "positive": positive,
            "negative": negative,
        }
    })
```

Note: The pipeline with top_k=None returns results sorted by score descending. When the model returns both labels, the sum of scores may not be exactly 1.0 but close dashboards.
