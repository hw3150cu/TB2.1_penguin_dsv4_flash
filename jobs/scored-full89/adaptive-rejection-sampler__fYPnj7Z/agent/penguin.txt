```
sample_envelope <- function(hull, uv = runif(2)) {
  log_areas <- hull$log_areas
  m <- length(log_areas)
  # log-sum-exp
  M <- max(log_areas)
  w <- exp(log_areas - M)
  total <- sum(w)
  u1 <- uv[1]
  # select interval
  target <- u1 * total
  cs <- cumsum(w)
  idx <- which(cs >= target)[1]
  if (is.na(idx)) idx <- m
  u2 <- min(max(uv[2], .Machine$double.eps), 1 - .Machine$double.eps)
  x <- sample_segment(hull, idx, u2)

}
```

```
sample_segment <- function(hull, i, u) {
  b <- hull$b[i]; s <- hull$s[i]; xi <- hull$x[i]; l <- hull$z[i]; r <- hull$z[i+1]
  if (!is.finite(l) && !is.finite(r)) stop
  if (is.infinite(l)) {
     if (s <= 0) stop("invalid left-unbounded segment")
     return(r + log(u)/s)
  }
  if (is.infinite(r)) {
     if (s >= 0) stop("invalid right-unbounded segment")
     return(l + log(1 - u)/s)
  }
  if (abs(s) < 1e-300) return(l + u*(r-l))
  A <- b + s*(l - xi); B <- b + s*(r - xi)
  M <- max(A, B)
  if (!is.finite(M)) stop("overflow in envelope sampling")
  alpha <- exp(A-M)*(1-u) + exp(B-M)*u
  # To avoid alpha=0 due underflow when u=0 or 1
  alpha <- max(alpha, .Machine$double.eps)
  t <- xi + (M + log(alpha) - b)/s
  # clamp
  t <- max(t, l); t <- min(t, r)
  t
}
```

```
numeric_gradient <- function(density_function, x, lower, upper) {
  # vectorized
  n <- length(x)
  h <- 1e-6 * pmax(abs(x), 1)
  central <- (x - h > lower) & (x + h < upper)
  hplus <- pmin(h, upper - x)
  hminus <- pmin(h, x - lower)
  # for infinite bounds pmax/min works
  # Forward/backward
  eps <- 1e-6 * pmax(abs(x), 1)
  deriv <- numeric(n)
  for (i in seq_len(n)) {
    if (central[i]) {
       deriv[i] <- (log_density(density_function, x[i]+eps[i]) - log_density(density_function, x[i]-eps[i])) / (2*eps[i])
    } else if (x[i] - lower <= upper - x[i]) {
       # closer to lower or right infinite; use forward
       hstep <- max(min(eps[i]*2, upper - x[i]), .Machine$double.eps)
       deriv[i] <- (log_density(density_function, x[i]+hstep) - log_density(density_function, x[i])) / hstep
    } else {
       hstep <- max(min(eps[i]*2, x[i] - lower), .Machine$double.eps)
       deriv[i] <- (log_density(density_function, x[i]) - log_density(density_function, x[i]-hstep)) / hstep
    }
  }
  deriv
}
```
For x at boundary exactly (x==lower), `x[i]-lower=0`, `x[i]-lower <= upper-x` likely true, so forward. hstep min(2eps, upper-x). Good.
For x near lower but density at lower is -Inf and central false, forward uses logd(x) finite and logd(x+hstep) finite, fine.

For infinite lower, `x - h > -Inf` is TRUE, central condition uses `x + h < upper` (if upper finite). For normal both infinities, central true.
