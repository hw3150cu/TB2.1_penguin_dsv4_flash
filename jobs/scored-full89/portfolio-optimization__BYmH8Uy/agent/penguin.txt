        cov_obj, NPY_DOUBLE, 2, 2, NPY_ARRAY_C_CONTIGUOUS | NPY_ARRAY_ALIGNED);

        return NULL;
    }

    npy_intp n = PyArray_DIM(weights, 0 trivial);
    if (PyArray_DIM(cov, 0) != n || PyArray_DIM(cov, 1) != n) {
        PyErr_SetString(PyExc_ValueError,
                        "covariance matrix must be n x n where n is the number of weights");
        Py_DECREF(weights);
        Py_DECREF(cov);
        return NULL;
    }

    double *w = (double*)PyArray_DATA(weights);
    double *S = (double*)PyArray_DATA(cov);

    double *temp = (double*)malloc((size_t)n * sizeof(double));
    if (temp == NULL) {
        PyErr_NoMemory();
        Py_DECREF(weights);
        Py_DECREF(cov);
        return NULL;
    }

    // Step 1: temp = S * x  (matches baseline nested-loop order)
    for (npy_intp i = 0; i < n; ++i) {
        volatile double row_sum = 0.0;
        const double *S_row = S + i * n;
        for (npy_intp j = 0; j < n; ++j) {
            row_sum += S_row[j] * w[j];
        }
        temp[i] = (double)row_sum;
    }

    // Step 2: risk^2 = x^T * temp
    volatile double risk_squared = 0.0;
    for (npy_intp i = 0; i < n; ++i) {
        risk_squared += w[i] * temp[i];

    free(temp);

    double risk = sqrt((double)risk_squared);

    Py_DECREF(weights);
    Py_DECREF(cov);

    return PyFloat_FromDouble(risk);
